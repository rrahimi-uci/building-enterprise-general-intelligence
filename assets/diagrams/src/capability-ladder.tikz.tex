% Figure 1-2 -- The four levels from a model to an enterprise agentic system.
% Restyled from capability-ladder.gv; columns and rows are unchanged. The
% level-three failure was corrected in v1.0.0 review: "runs forever; loses
% everything on restart" overstated it, since frameworks cap turns and can
% persist sessions and run state (Chapter 1, Level three).
%
% Original element -> restyled element
%   h1 LEVEL | h2 WHAT IT ADDS | h3 THE NEW FAILURE CLASS
%        -> three column headings (navy, teal, red text) each over a hairline rule
%   r1..r4  "1 . Model", "2 . Model + tools", "3 . Agent", "4 . Enterprise system"
%        -> four level cards (\dgrow, a local macro): the number is the solid
%           badge, the name the bold title; hues slate / cyan / teal / blue
%   a1..a4  language understanding | contact with current reality |
%           adaptive multi-step work | durability . authority, evidence . control
%        -> four description cards, same hue as their level
%   f1..f4  confidently wrong about facts it cannot reach | acts on data the
%           caller was never authorized to see | stops only at a turn limit;
%           a restart loses state the runtime did not save | complexity -- which
%           is why the boundaries are explicit
%        -> four failure cards: red, red, red and amber (the source's fourth is
%           amber)
%   edges: none (the source rows and columns are placed by invisible chains)
\documentclass[border=6pt]{standalone}
\usepackage{professional-diagrams}

% ---- local macro (candidate for promotion to professional-diagrams.sty) ----
% \dgrow[opts]{name}{role}{(x,y)}{badge}{content}: a row card with a solid
% badge (number / numeral) in the role's border colour, inset at the left, and
% text indented to clear it. \dgrowh = row height, \dgrowbw = badge width.
\newlength{\dgrowh}  \setlength{\dgrowh}{52pt}
\newlength{\dgrowbw} \setlength{\dgrowbw}{24pt}
\newcommand{\dgrow}[6][]{%
  \node[dg card=#3, minimum height=\dgrowh, text width=475pt,
        execute at begin node={\leftskip=\dimexpr\dgrowbw+7pt\relax}, #1] (#2) at #4 {#6};
  \node[rectangle, rounded corners=4pt, fill=dg-#3-line, text=white, font=\dgtitlesize,
        inner sep=0pt, minimum width=\dgrowbw, minimum height=\dgrowh-14pt, anchor=west]
    at ([xshift=7pt]#2.west) {#5};}

\begin{document}
\begin{tikzpicture}
  % ---- column headings ----
  \foreach \x/\w/\r/\t in {80.5/161/navy/LEVEL, 242.5/139/teal/WHAT IT ADDS, 408/168/red/THE NEW FAILURE CLASS} {
    \node[anchor=south west, inner sep=0pt, text=dg-\r-ink, font=\dggroupsize]
      at (\x-\w/2+2,42) {\t};
    \draw[dg-\r-line, line width=1pt, line cap=round] (\x-\w/2,38) -- (\x+\w/2,38);
  }

  % ---- rows ----
  \foreach \i/\role/\lvl/\name/\add/\fr/\frrole [evaluate=\i as \y using {-\i*66+0}] in {
      0/slate/1/Model/{language\\understanding}/{confidently wrong about\\facts it cannot reach}/red,
      1/cyan/2/Model + tools/{contact with\\current reality}/{acts on data the caller\\was never authorized to see}/red,
      2/teal/3/Agent/{adaptive\\multi-step work}/{stops only at a turn limit;\\a restart loses state\\the runtime did not save}/red,
      3/blue/4/Enterprise system/{durability · authority\\evidence · control}/{complexity — which is why\\the boundaries are explicit}/amber} {
    \dgrow[text width=140pt]{lv\i}{\role}{(80.5,\y)}{\lvl}{\dgcontent{\name}{}{}}
    \node[dg card=\role, text width=118pt, minimum height=52pt] (ad\i) at (242.5,\y)
      {\dgcontent{}{\add}{}};
    \node[dg card=\frrole, text width=147pt, minimum height=52pt] (fr\i) at (408,\y)
      {\dgcontent{}{\fr}{}};
  }
\end{tikzpicture}
\end{document}
