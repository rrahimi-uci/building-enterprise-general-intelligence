%   refuse  Refuse or escalate                              -> red card
%   cheap   Cheapest approved model that meets the bar      -> teal card
%   strong  Higher-capability approved model                -> teal card
%   edges   req>g1 (plain), g1>g2 "yes", g2>g3 "yes" (blue),

%           g3>cheap "yes" (teal), g3>strong "no" (blue)
\documentclass[border=6pt]{standalone}
\usepackage{professional-diagrams}
% local: a two-line bold question inside a diamond (each line sized on its own,
% because align=center restarts the font after a line break)
\newcommand{\dgq}[2]{{\dgtitlesize #1}\\{\dgtitlesize #2}}
\begin{document}
\begin{tikzpicture}
  % ---- spine: request and the three questions ----
  \node[dg card=slate, text width=170pt] (req) at (130,0)
    {\dgcontent{Request}{purpose · data class · region}{}};
  \node[dg decision=red, minimum width=215pt, minimum height=76pt] (g1) at (130,-105)
    {\dgq{Residency and retention}{permit this provider?}};
  \node[dg decision=red, minimum width=215pt, minimum height=76pt] (g2) at (130,-217)
    {\dgq{Model approved for}{this purpose?}};
  \node[dg decision=amber, minimum width=215pt, minimum height=76pt] (g3) at (130,-329)
    {\dgq{Capability sufficient}{for the task?}};

  % ---- outcomes ----
  \node[dg card=red, text width=150pt] (refuse) at (410,-217)
    {\dgcontent{Refuse or escalate}{}{}};
  \node[dg card=teal, text width=150pt] (cheap) at (130,-436)
    {\dgcontent{Cheapest approved model}{that meets the bar}{}};
  \node[dg card=teal, text width=150pt] (strong) at (410,-329)
    {\dgcontent{Higher-capability approved model}{}{}};

  % ---- connectors ----
  \draw[flow in=blue] (req.south) -- (g1.north);
  \draw[flow in=blue] (g1.south) -- node[pill=blue]{Yes} (g2.north);
  \draw[flow in=blue] (g2.south) -- node[pill=blue]{Yes} (g3.north);
  \draw[flow in=red, turn] (g1.east) -| node[pill=red, pos=0.2]{No} (refuse.north);
  \draw[flow in=red] (g2.east) -- node[pill=red]{No} (refuse.west);
  \draw[flow in=teal] (g3.south) -- node[pill=teal]{Yes} (cheap.north);
  \draw[flow in=blue] (g3.east) -- node[pill=blue]{No} (strong.west);
\end{tikzpicture}
\end{document}
