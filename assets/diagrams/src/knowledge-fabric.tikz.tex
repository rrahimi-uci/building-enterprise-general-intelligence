% Local macros (candidates for promotion): a row cell and an untitled row panel.
\newcommand{\dgrowcell}[5]{% name, role, x,y, content
  \node[dg card=#2, text width=92pt, inner xsep=7pt, minimum height=44pt, align=center] (#1) at (#3,#4) {#5};}
\newcommand{\dgrowpanel}[3]{% name, role, fit list

    \node[fit=#3, inner sep=7pt, rounded corners=9pt, fill=white, draw=dg-#2-line!45,
          line width=0.8pt] (#1) {};
  \end{pgfonlayer}}
